    function computeProgress(goal){

      if(goal.uomType === 'numeric-min') return Math.min(100, Math.round((Number(goal.actual||0) / Number(goal.target||1)) * 100));

      if(goal.uomType === 'numeric-max') return Number(goal.actual||0) <= 0 ? 0 : Math.min(100, Math.round((Number(goal.target||0) / Number(goal.actual||1)) * 100));

      if(goal.uomType === 'timeline') return String(goal.actual || '') <= String(goal.target || '') ? 100 : 50;

      if(goal.uomType === 'zero') return Number(goal.actual||0) === 0 ? 100 : 0;

      return 0;

    }


    function statusClass(status){

      if(status === 'Completed') return 's-done';

      if(status === 'On Track') return 's-track';

      if(status === 'Locked') return 's-lock';

      if(status === 'At Risk') return 's-risk';

      return 's-draft';

    }


    function flash(id, type, msg){

      const el = q(id); if(!el) return; el.className = 'alert ' + type + ' show'; el.textContent = msg;

      setTimeout(() => el.classList.remove('show'), 2600);

    }


    function audit(user, action, details){

      const now = new Date();

      const stamp = now.getFullYear() + '-' + String(now.getMonth()+1).padStart(2,'0') + '-' + String(now.getDate()).padStart(2,'0') + ' ' + String(now.getHours()).padStart(2,'0') + ':' + String(now.getMinutes()).padStart(2,'0');

      state.audit.unshift({when: stamp, user, action, details});

    }

      q('#kpiProgress').textContent = avg + '%';

      q('#kpiCycle').textContent = state.cycleState;

          <div class="stack">

            <label class="small muted">Target</label>

            <input ${g.shared ? targetReadOnly : ''} ${targetDisabled} class="goal-target" value="${g.target}">

          </div>

          <div class="stack">

            <label class="small muted">Weight %</label>

            <input ${weightDisabled} class="goal-weight" type="number" min="0" max="100" value="${g.weightage}">

          </div>

          <div class="stack">

            <label class="small muted">Action</label>

            <button class="btn remove-goal" ${readOnly || managerMode || g.shared ? 'disabled' : ''}>Remove</button>

          </div>

          <div class="col-12 stack">

            <label class="small muted">Description</label>

            <textarea ${descDisabled} class="goal-desc">${g.description}</textarea>

          </div>

        </div>`;

    }


    function syncGoalsFromUI(){

      qa('.goal-row').forEach(row => {

        const id = Number(row.dataset.id);

        const g = state.goals.find(x => x.id === id);

        if(!g) return;

        g.title = row.querySelector('.goal-title').value;

        g.thrust = row.querySelector('.goal-thrust').value;

        g.uomType = row.querySelector('.goal-uom').value;

        g.target = row.querySelector('.goal-target').value;

        g.weightage = Number(row.querySelector('.goal-weight').value || 0);

        g.description = row.querySelector('.goal-desc').value;

      });

      const shared = state.goals.find(g => g.shared);

      if(shared) shared.actual = state.sharedMasterAchievement;

    }


    function validateGoals(){

      syncGoalsFromUI();

      if(state.goals.length > 8) return 'Maximum 8 goals are allowed.';

      const low = state.goals.find(g => Number(g.weightage||0) < 10);

      if(low) return 'Each goal must carry at least 10% weightage.';

      const total = state.goals.reduce((a,g)=>a+Number(g.weightage||0),0);

      if(total !== 100) return 'Total weightage across all goals must equal 100%.';

      return '';

    }
